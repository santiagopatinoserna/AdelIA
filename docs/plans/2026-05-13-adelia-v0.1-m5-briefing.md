# AdelIA v0.1 — M5 Briefing Generator Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship Modo B briefing for medical referrals. User selects target specialty (dropdown 8+default); backend async-generates a 1-page SBAR-shaped briefing with 8 universal sections filtered by specialty rules + top-5 pre-baked Q&A with verbatim citations. Frontend polls until ready, renders Jinja2 SSR HTML with click-to-source citation footnotes.

**Architecture:** Five new backend modules form the briefing path: `sbar_formatter` (pure, entities → 8 sections), `qa_pre_baker` (1 LLM call with anti-hallucination guard), `briefing_html_renderer` (Jinja2 + nh3 sanitization), `briefing_repo` (CRUD), `briefing_orchestrator` (5-stage pipeline coordinator). New router `briefings.py` exposes POST/GET/list/regenerate. Frontend adds 3 pages under `/dashboard/briefings/` plus a sanitized-HTML render component. Reuses M2 entity model + M3 chat_orchestrator mocked-test pattern + M2 TDD leaf worker pattern + M1 BackgroundTasks pattern. Briefing Jinja2 template is shared with M6 WeasyPrint PDF (single source of truth).

**Tech Stack:**
- **Backend additions:** `jinja2>=3.1` (pinned), `nh3>=0.2` (Rust-backed HTML sanitizer)
- **Reuses:** FastAPI, SQLAlchemy 2.0 async, Alembic, Pydantic v2, mistralai 2.4.5 (`from mistralai.client.sdk import Mistral`), OpenAI 2.36 (used only for Whisper in M2, not M5)
- **Frontend additions:** none — uses existing Next.js 15 / TS 5.9 / Tailwind 4 / next-intl v3 / Auth.js v5

**Branch:** `feat/m5-briefing` (cut from `feat/m3-rag-chat` head, which contains M1+M2+M3).

```python
# ============================================================================
# M5: Briefing schemas

from enum import Enum


class Specialty(str, Enum):
    PEDIATRICS = "pediatrics"
    ENDOCRINOLOGY = "endocrinology"
    CARDIOLOGY = "cardiology"
    GYNECOLOGY = "gynecology"
    GASTROENTEROLOGY = "gastroenterology"
    DERMATOLOGY = "dermatology"
    GENERAL_MEDICINE = "general_medicine"
    DEFAULT = "default"


class BriefingStatus(str, Enum):
    PROCESSING = "processing"
    READY = "ready"
    FAILED = "failed"


class Citation(BaseModel):
    document_id: UUID
    entity_id: UUID | None = None
    raw_quote: str = Field(min_length=1, max_length=2000)
    page: int | None = None


class ProblemItem(BaseModel):
    name: str
    status: str | None = None  # active | resolved | unknown
    onset_date: str | None = None
    citations: list[Citation] = Field(default_factory=list)
    provenance: Literal["extracted_from_clinician_doc", "patient_stated"]


class MedicationItem(BaseModel):
    name: str
    dose: str | None = None
    route: str | None = None
    frequency: str | None = None
    started: str | None = None
    citations: list[Citation] = Field(default_factory=list)
    provenance: Literal["extracted_from_clinician_doc", "patient_stated"]


class AllergyItem(BaseModel):
    substance: str
    reaction: str | None = None
    severity: str | None = None
    citations: list[Citation] = Field(default_factory=list)
    provenance: Literal["extracted_from_clinician_doc", "patient_stated"]


class ImmunizationItem(BaseModel):
    name: str
    date: str | None = None
    citations: list[Citation] = Field(default_factory=list)
    provenance: Literal["extracted_from_clinician_doc", "patient_stated"]


class ProcedureItem(BaseModel):
    name: str
    date: str | None = None
    citations: list[Citation] = Field(default_factory=list)
    provenance: Literal["extracted_from_clinician_doc", "patient_stated"]


class FamilyHistoryItem(BaseModel):
    relation: str
    condition: str
    citations: list[Citation] = Field(default_factory=list)
    provenance: Literal["extracted_from_clinician_doc", "patient_stated"]


class ObservationItem(BaseModel):
    name: str           # 'HbA1c', 'BP', 'glucose'
    value: str
    unit: str | None = None
    date: str | None = None
    trend: str | None = None  # 'rising' | 'falling' | 'stable' | None
    citations: list[Citation] = Field(default_factory=list)
    provenance: Literal["extracted_from_clinician_doc", "patient_stated"]


class QAItem(BaseModel):
    question: str
    answer: str
    citations: list[Citation]


class BriefingSections(BaseModel):
    reason_for_consult: str | None
    active_problems: list[ProblemItem]
    medications: list[MedicationItem]
    allergies: list[AllergyItem] | Literal["no_known_allergies"]
    immunizations: list[ImmunizationItem] | None  # None = hidden by specialty filter
    procedures: list[ProcedureItem]
    family_history: list[FamilyHistoryItem] | None  # None = hidden
    relevant_observations: list[ObservationItem]
    open_questions: list[QAItem]


class PatientHeader(BaseModel):
    name: str
    age: int
    sex: str


class AppliedFilter(BaseModel):
    specialty: str
    rules_version: str


class LLMMeta(BaseModel):
    model: str
    tokens_used: int
    cost_estimate_usd: float


class BriefingPayload(BaseModel):
    schema_version: Literal["1.0"] = "1.0"
    generated_at: datetime
    target_specialty: Specialty
    patient: PatientHeader
    sections: BriefingSections
    applied_filter: AppliedFilter
    llm_meta: LLMMeta


class BriefingCreate(BaseModel):
    target_specialty: Specialty


class BriefingOut(BaseModel):
    id: UUID
    target_specialty: Specialty
    status: BriefingStatus
    is_stale: bool
    created_at: datetime
    completed_at: datetime | None = None
    error_message: str | None = None


class BriefingDetailOut(BriefingOut):
    payload: BriefingPayload | None = None
    html_render: str | None = None
```

```python
"""Loader for specialty filter rule JSON templates."""
from __future__ import annotations

import json
from pathlib import Path

from pydantic import BaseModel, Field

from adelia.schemas import Specialty


class SpecialtyTemplate(BaseModel):
    specialty: str
    rules_version: str
    include_immunizations: bool
    include_family_history: bool
    include_social_history: bool = False
    highlight_observations: list[str] = Field(default_factory=list)
    exclude_observations: list[str] = Field(default_factory=list)
    vital_emphasis: list[str] = Field(default_factory=list)
    qa_pre_baker_hints: str


_TEMPLATES_DIR = Path(__file__).parent


def load_template(specialty: Specialty) -> SpecialtyTemplate:
    """Load a SpecialtyTemplate by enum value. Raises FileNotFoundError if missing."""
    path = _TEMPLATES_DIR / f"{specialty.value}.json"
    with path.open("r", encoding="utf-8") as f:
        data = json.load(f)
    return SpecialtyTemplate(**data)


def list_specialties() -> set[str]:
    """Return the set of specialty filenames present (without .json extension)."""
    return {p.stem for p in _TEMPLATES_DIR.glob("*.json")}
```

```python
"""Pure entity-to-section formatter for briefings.

NO DB, NO LLM, NO API. Takes a list of EntityInput + specialty + optional
reason_for_consult; returns a BriefingSections payload with the 8 universal
sections, filtered per specialty filter rules.

Trade-off: this is intentionally NOT a service that loads from DB. The orchestrator
loads entities first (with the user_id + deleted_at filter), then passes them here
as a pure transformation. Keeps this module testable without Postgres.
"""
from __future__ import annotations

from typing import Literal
from uuid import UUID

from pydantic import BaseModel, Field

from adelia.schemas import (
    AllergyItem,
    BriefingSections,
    Citation,
    FamilyHistoryItem,
    ImmunizationItem,
    MedicationItem,
    ObservationItem,
    ProblemItem,
    ProcedureItem,
    Specialty,
)
from adelia.specialty_templates import SpecialtyTemplate, load_template


class EntityInput(BaseModel):
    """Subset of Entity ORM fields needed by sbar_formatter.

    Decoupled from ORM so the formatter remains pure.
    """
    id: UUID
    document_id: UUID
    entity_type: str
    name: str
    value: dict = Field(default_factory=dict)
    event_date: str | None = None
    raw_quote: str
    provenance: Literal["extracted_from_clinician_doc", "patient_stated"]


def _to_citation(e: EntityInput) -> Citation:
    return Citation(
        document_id=e.document_id,
        entity_id=e.id,
        raw_quote=e.raw_quote,
    )


def _filter_observations(
    entities: list[EntityInput], template: SpecialtyTemplate
) -> list[EntityInput]:
    """Apply specialty whitelist/blacklist to observation entities.

    Empty whitelist = include all (subject to blacklist).
    """
    obs = [e for e in entities if e.entity_type == "observation"]

    if template.highlight_observations:
        whitelist = set(template.highlight_observations)
        obs = [e for e in obs if e.name in whitelist]

    if template.exclude_observations:
        blacklist = set(template.exclude_observations)
        obs = [e for e in obs if e.name not in blacklist]

    return obs


def format_sections(
    entities: list[EntityInput],
    specialty: Specialty,
    reason_for_consult: str | None,
) -> BriefingSections:
    """Transform a list of entities into a BriefingSections payload.

    Applies specialty filter rules. Idempotent and deterministic given
    the same inputs.
    """
    template = load_template(specialty)

    # Group entities by type
    by_type: dict[str, list[EntityInput]] = {}
    for e in entities:
        by_type.setdefault(e.entity_type, []).append(e)

    # Active problems (condition entities)
    active_problems = [
        ProblemItem(
            name=e.name,
            status=e.value.get("status") if isinstance(e.value, dict) else None,
            onset_date=e.event_date,
            citations=[_to_citation(e)],
            provenance=e.provenance,
        )
        for e in by_type.get("condition", [])
    ]

    # Medications
    medications = [
        MedicationItem(
            name=e.name,
            dose=e.value.get("dose") if isinstance(e.value, dict) else None,
            route=e.value.get("route") if isinstance(e.value, dict) else None,
            frequency=e.value.get("frequency") if isinstance(e.value, dict) else None,
            started=e.event_date,
            citations=[_to_citation(e)],
            provenance=e.provenance,
        )
        for e in by_type.get("medication", [])
    ]

    # Allergies — "No known allergies" if empty (IPS convergent kernel rule)
    allergy_entities = by_type.get("allergy", [])
    if allergy_entities:
        allergies: list[AllergyItem] | str = [
            AllergyItem(
                substance=e.name,
                reaction=e.value.get("reaction") if isinstance(e.value, dict) else None,
                severity=e.value.get("severity") if isinstance(e.value, dict) else None,
                citations=[_to_citation(e)],
                provenance=e.provenance,
            )
            for e in allergy_entities
        ]
    else:
        allergies = "no_known_allergies"

    # Immunizations — only if specialty rule says so
    immunizations: list[ImmunizationItem] | None = None
    if template.include_immunizations:
        immunizations = [
            ImmunizationItem(
                name=e.name,
                date=e.event_date,
                citations=[_to_citation(e)],
                provenance=e.provenance,
            )
            for e in by_type.get("immunization", [])
        ]

    # Procedures
    procedures = [
        ProcedureItem(
            name=e.name,
            date=e.event_date,
            citations=[_to_citation(e)],
            provenance=e.provenance,
        )
        for e in by_type.get("procedure", [])
    ]

    # Family history — only if specialty rule says so
    family_history: list[FamilyHistoryItem] | None = None
    if template.include_family_history:
        fh_entities = by_type.get("family_history", [])
        family_history = [
            FamilyHistoryItem(
                relation=e.value.get("relation", "unknown") if isinstance(e.value, dict) else "unknown",
                condition=e.name,
                citations=[_to_citation(e)],
                provenance=e.provenance,
            )
            for e in fh_entities
        ]

    # Observations — filtered by specialty
    obs_filtered = _filter_observations(entities, template)
    relevant_observations = [
        ObservationItem(
            name=e.name,
            value=str(e.value.get("value", "")) if isinstance(e.value, dict) else "",
            unit=e.value.get("unit") if isinstance(e.value, dict) else None,
            date=e.event_date,
            trend=None,  # M5: no trend calc. v0.1.1 enhancement.
            citations=[_to_citation(e)],
            provenance=e.provenance,
        )
        for e in obs_filtered
    ]

    return BriefingSections(
        reason_for_consult=reason_for_consult,
        active_problems=active_problems,
        medications=medications,
        allergies=allergies,
        immunizations=immunizations,
        procedures=procedures,
        family_history=family_history,
        relevant_observations=relevant_observations,
        open_questions=[],  # qa_pre_baker fills this in the orchestrator
    )
```

```python
logger = logging.getLogger(__name__)


class QAPrebakerError(Exception):
    """Raised when the Mistral call fails irrecoverably."""


_QA_JSON_SCHEMA = {
    "type": "object",
    "properties": {
        "items": {
            "type": "array",
            "minItems": 1,
            "maxItems": 5,
            "items": {
                "type": "object",
                "properties": {
                    "question": {"type": "string", "minLength": 1},
                    "answer": {"type": "string", "minLength": 1},
                    "citations": {
                        "type": "array",
                        "minItems": 1,
                        "items": {
                            "type": "object",
                            "properties": {
                                "document_id": {"type": "string"},
                                "entity_id": {"type": ["string", "null"]},
                                "raw_quote": {"type": "string", "minLength": 1},
                            },
                            "required": ["document_id", "raw_quote"],
                        },
                    },
                },
                "required": ["question", "answer", "citations"],
            },
        }
    },
    "required": ["items"],
}


def _system_prompt(specialty: Specialty, hints: str) -> str:
    return (
        "You are a clinical assistant generating pre-baked Q&A for a medical briefing.\n\n"
        f"You are preparing a briefing for a patient about to see a {specialty.value} specialist.\n"
        "Generate exactly 5 questions a specialist is most likely to ask in a 15-minute consultation, "
        "based ONLY on the patient data provided below. For each question, provide an answer.\n\n"
        "CRITICAL ANTI-HALLUCINATION RULES:\n"
        "- If patient data does not contain enough info to answer a question, answer 'No data available in patient history'.\n"
        "- DO NOT invent diagnoses, medications, observations, or values not present in the entities.\n"
        "- Each answer MUST cite at least one raw_quote from the source documents — verbatim, no paraphrasing.\n"
        "- raw_quote MUST appear textually in one of the provided source quotes (no rephrasing, no synthesizing).\n"
        "- If you cannot generate 5 questions with grounded answers, generate fewer (minimum 3) — never fabricate.\n\n"
        f"Specialty-specific guidance: {hints}\n\n"
        "Output JSON matching the provided schema."
    )


def _build_client():
    from mistralai.client.sdk import Mistral

    if not settings.mistral_api_key:
        raise QAPrebakerError("MISTRAL_API_KEY not configured")
    return Mistral(api_key=settings.mistral_api_key)


def _llm_structured_call(
    client: Any,
    system_prompt: str,
    user_content: str,
    model: str,
) -> tuple[dict, int]:
    """Call Mistral chat.complete with structured output. Returns (parsed_dict, tokens_used)."""
    from mistralai.client.errors import MistralError

    try:
        resp = client.chat.complete(
            model=model,
            messages=[
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": user_content},
            ],
            response_format={
                "type": "json_schema",
                "json_schema": {"name": "qa_items", "schema": _QA_JSON_SCHEMA, "strict": True},
            },
            temperature=0.2,
        )
    except MistralError as e:
        raise QAPrebakerError(f"Mistral API error: {e}") from e
    except Exception as e:
        raise QAPrebakerError(f"Unexpected error during Mistral call: {e}") from e

    content = resp.choices[0].message.content
    tokens_used = getattr(resp.usage, "total_tokens", 0) if getattr(resp, "usage", None) else 0
    try:
        data = json.loads(content)
    except json.JSONDecodeError as e:
        raise QAPrebakerError(f"Failed to parse JSON from LLM: {e}") from e
    return data, tokens_used


def _enforce_raw_quote(items: list[dict], source_raw_texts: dict[str, str]) -> list[QAItem]:
    """Drop items whose citations don't appear verbatim in any source doc.

    Defense against LLM hallucination. raw_quote MUST be a substring of some doc's raw_text.
    """
    all_text = " \n ".join(source_raw_texts.values())
    kept: list[QAItem] = []
    for item_dict in items:
        valid_cits: list[Citation] = []
        for cit in item_dict.get("citations", []):
            quote = cit.get("raw_quote", "")
            if quote and quote in all_text:
                try:
                    doc_id = UUID(cit["document_id"])
                except (ValueError, KeyError):
                    logger.warning("Invalid document_id in citation, skipping: %r", cit)
                    continue
                entity_id_raw = cit.get("entity_id")
                entity_id: UUID | None
                if entity_id_raw:
                    try:
                        entity_id = UUID(entity_id_raw)
                    except ValueError:
                        entity_id = None
                else:
                    entity_id = None
                valid_cits.append(
                    Citation(document_id=doc_id, entity_id=entity_id, raw_quote=quote)
                )
            else:
                logger.warning(
                    "Anti-hallucination: dropping citation, raw_quote not in source: %r",
                    quote,
                )
        if not valid_cits:
            logger.warning(
                "Anti-hallucination: dropping QA item with no valid citations: %r",
                item_dict.get("question"),
            )
            continue
        try:
            kept.append(
                QAItem(
                    question=item_dict["question"],
                    answer=item_dict["answer"],
                    citations=valid_cits,
                )
            )
        except (ValidationError, KeyError) as e:
            logger.warning("Anti-hallucination: dropping malformed QA item: %s", e)
            continue
    return kept


def bake_questions(
    specialty: Specialty,
    entities_text: str,
    specialty_hints: str,
    source_raw_texts: dict[str, str],
    model: str = "mistral-medium-latest",
) -> tuple[list[QAItem], LLMMeta]:
    """Generate top-5 Q&A from patient context, with anti-hallucination filter.

    Returns (items, meta). Items may be 0-5 after filter; LLMMeta records cost/tokens.
    """
    client = _build_client()
    system_prompt = _system_prompt(specialty, specialty_hints)
    user_content = (
        "Patient data (entities + raw_quotes):\n\n"
        f"{entities_text}\n\n"
        "Source documents (for verbatim citation lookup):\n\n"
        + "\n\n".join(f"--- {doc_id} ---\n{text[:4000]}" for doc_id, text in source_raw_texts.items())
    )

    parsed, tokens_used = _llm_structured_call(client, system_prompt, user_content, model)
    items_raw = parsed.get("items", [])
    items = _enforce_raw_quote(items_raw, source_raw_texts)

    # Cost estimate: mistral-medium-latest ~ $2.7/1M input tokens + $8.1/1M output.
    # Rough estimate: assume 50% in, 50% out → blended ~$5.4/1M.
    cost_estimate_usd = (tokens_used / 1_000_000) * 5.4

    meta = LLMMeta(model=model, tokens_used=tokens_used, cost_estimate_usd=cost_estimate_usd)
    return items, meta
```

```python
"""Briefing HTML renderer — Jinja2 server-side render + nh3 sanitization.

The same template is reused by M6 WeasyPrint PDF render (single source of truth).
nh3 sanitizes LLM-generated text fields BEFORE templating, not after, to prevent
HTML injection via answers/narrative strings.
"""
from __future__ import annotations

import logging
from pathlib import Path

import nh3
from jinja2 import Environment, FileSystemLoader, select_autoescape

from adelia.schemas import BriefingPayload

logger = logging.getLogger(__name__)


_TEMPLATE_DIR = Path(__file__).parent.parent / "templates"
_NH3_ALLOWED_TAGS: set[str] = set()  # strip ALL HTML from LLM text
_NH3_ALLOWED_ATTRS: dict[str, set[str]] = {}


def _make_env() -> Environment:
    env = Environment(
        loader=FileSystemLoader(str(_TEMPLATE_DIR)),
        autoescape=select_autoescape(["html", "j2"]),
        trim_blocks=True,
        lstrip_blocks=True,
    )
    return env


def _sanitize_payload(payload: BriefingPayload) -> BriefingPayload:
    """Run nh3.clean over every user-visible string field that could carry LLM HTML."""
    p = payload.model_copy(deep=True)

    if p.sections.reason_for_consult:
        p.sections.reason_for_consult = nh3.clean(
            p.sections.reason_for_consult,
            tags=_NH3_ALLOWED_TAGS,
            attributes=_NH3_ALLOWED_ATTRS,
        )

    for qa in p.sections.open_questions:
        qa.question = nh3.clean(qa.question, tags=_NH3_ALLOWED_TAGS, attributes=_NH3_ALLOWED_ATTRS)
        qa.answer = nh3.clean(qa.answer, tags=_NH3_ALLOWED_TAGS, attributes=_NH3_ALLOWED_ATTRS)
        for c in qa.citations:
            c.raw_quote = nh3.clean(
                c.raw_quote, tags=_NH3_ALLOWED_TAGS, attributes=_NH3_ALLOWED_ATTRS
            )

    return p


def render_briefing_html(payload: BriefingPayload) -> str:
    """Render the briefing payload to a complete HTML document.

    Args:
        payload: validated BriefingPayload.
    Returns:
        Full HTML string (DOCTYPE + html + head + body).
    Raises:
        jinja2.TemplateError if template syntax fails.
    """
    sanitized = _sanitize_payload(payload)
    env = _make_env()
    template = env.get_template("briefing/briefing.html.j2")
    html = template.render(payload=sanitized)
    return html
```

```python
"""briefing_repo — CRUD over Briefing + BriefingCitation tables.

Cross-user isolation enforced at every read: WHERE user_id = :uid AND deleted_at IS NULL.
"""
from __future__ import annotations

from datetime import datetime, UTC
from uuid import UUID

from sqlalchemy import select, update
from sqlalchemy.ext.asyncio import AsyncSession

from adelia.models import Briefing
from adelia.schemas import BriefingPayload, Specialty


async def create_briefing(
    session: AsyncSession,
    *,
    user_id: UUID,
    target_specialty: Specialty,
) -> Briefing:
    """Insert a new briefing row with status='processing'."""
    b = Briefing(
        user_id=user_id,
        target_specialty=target_specialty.value,
        status="processing",
        is_stale=False,
    )
    session.add(b)
    await session.flush()
    return b


async def get_by_id(
    session: AsyncSession,
    *,
    briefing_id: UUID,
    user_id: UUID,
) -> Briefing | None:
    """Return briefing if owned by user_id and not soft-deleted, else None (opaque 404)."""
    stmt = (
        select(Briefing)
        .where(Briefing.id == briefing_id)
        .where(Briefing.user_id == user_id)
        .where(Briefing.deleted_at.is_(None))
    )
    result = await session.execute(stmt)
    return result.scalar_one_or_none()


async def list_for_user(
    session: AsyncSession,
    *,
    user_id: UUID,
    limit: int = 50,
) -> list[Briefing]:
    """Return non-deleted briefings for user, newest first."""
    stmt = (
        select(Briefing)
        .where(Briefing.user_id == user_id)
        .where(Briefing.deleted_at.is_(None))
        .order_by(Briefing.created_at.desc())
        .limit(limit)
    )
    result = await session.execute(stmt)
    return list(result.scalars().all())


async def update_payload_html_status(
    session: AsyncSession,
    *,
    briefing_id: UUID,
    payload: BriefingPayload | None,
    html_render: str | None,
    status: str,
    error_message: str | None,
) -> None:
    """Update payload + html + status + completed_at + error_message."""
    stmt = (
        update(Briefing)
        .where(Briefing.id == briefing_id)
        .values(
            payload=payload.model_dump(mode="json") if payload else None,
            html_render=html_render,
            status=status,
            error_message=error_message,
            completed_at=datetime.now(UTC),
        )
    )
    await session.execute(stmt)


async def mark_stale_for_user(
    session: AsyncSession,
    *,
    user_id: UUID,
    since: datetime,
) -> int:
    """Mark all non-deleted briefings of user_id created before `since` as stale.

    Returns: count of rows updated. Idempotent (already-stale rows stay stale).
    """
    stmt = (
        update(Briefing)
        .where(Briefing.user_id == user_id)
        .where(Briefing.deleted_at.is_(None))
        .where(Briefing.created_at < since)
        .values(is_stale=True)
    )
    result = await session.execute(stmt)
    return result.rowcount or 0


async def soft_delete(
    session: AsyncSession,
    *,
    briefing_id: UUID,
    user_id: UUID,
) -> bool:
    """Soft-delete a briefing. Returns True if a row was updated."""
    stmt = (
        update(Briefing)
        .where(Briefing.id == briefing_id)
        .where(Briefing.user_id == user_id)
        .where(Briefing.deleted_at.is_(None))
        .values(deleted_at=datetime.now(UTC))
    )
    result = await session.execute(stmt)
    return (result.rowcount or 0) > 0
```

```python
"""Briefing orchestrator — 5-stage pipeline coordinator.

Stages:
  1. Load entities + source raw_texts (scoped to user_id, deleted_at IS NULL)
  2. sbar_formatter → BriefingSections.sections (8 sections, filtered)
  3. qa_pre_baker → top-5 QA + citations (with anti-hallucination filter)
  4. Assemble BriefingPayload
  5. briefing_html_renderer → HTML string
  finally: update briefing row with status (ready/failed) + payload + html + completed_at
"""
from __future__ import annotations

import logging
from datetime import datetime, UTC
from uuid import UUID

from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from adelia.db import AsyncSessionLocal
from adelia.models import Document, Entity, User
from adelia.schemas import (
    AppliedFilter,
    BriefingPayload,
    PatientHeader,
    Specialty,
)
from adelia.services import briefing_repo
from adelia.services.briefing_html_renderer import render_briefing_html
from adelia.services.qa_pre_baker import bake_questions, QAPrebakerError
from adelia.services.sbar_formatter import EntityInput, format_sections
from adelia.specialty_templates import load_template

logger = logging.getLogger(__name__)


async def _load_user_entities_and_sources(
    session: AsyncSession, user_id: UUID
) -> tuple[list[EntityInput], dict[str, str]]:
    """Load entities + source raw_texts for user, filtered by deleted_at IS NULL."""
    entity_stmt = (
        select(Entity)
        .join(Document, Entity.document_id == Document.id)
        .where(Document.user_id == user_id)
        .where(Document.deleted_at.is_(None))
        .where(Entity.deleted_at.is_(None))
        .limit(500)
    )
    result = await session.execute(entity_stmt)
    rows = result.scalars().all()

    entities = [
        EntityInput(
            id=e.id,
            document_id=e.document_id,
            entity_type=e.entity_type,
            name=e.name or "",
            value=e.value or {},
            event_date=e.event_date.isoformat() if e.event_date else None,
            raw_quote=e.raw_quote or "",
            provenance=e.provenance or "extracted_from_clinician_doc",
        )
        for e in rows
        if e.raw_quote  # skip entities lacking raw_quote (defensive)
    ]

    doc_ids = {e.document_id for e in entities}
    if doc_ids:
        doc_stmt = select(Document).where(Document.id.in_(doc_ids))
        doc_result = await session.execute(doc_stmt)
        sources = {str(d.id): (d.raw_text or "") for d in doc_result.scalars().all()}
    else:
        sources = {}

    return entities, sources


async def _load_user_patient_header(session: AsyncSession, user_id: UUID) -> PatientHeader:
    """Build a minimal PatientHeader from the User row. M5 placeholder until UserSettings is wired."""
    stmt = select(User).where(User.id == user_id)
    result = await session.execute(stmt)
    user = result.scalar_one_or_none()
    if user is None:
        # Defensive: should never happen if auth_dep enforced — but keep briefing alive.
        return PatientHeader(name="Unknown", age=0, sex="?")
    return PatientHeader(
        name=(user.email or "Unknown").split("@", 1)[0],
        age=0,
        sex="?",
    )


def _entities_to_text_block(entities: list[EntityInput]) -> str:
    """Format entities as a text block for the qa_pre_baker prompt."""
    lines = []
    for e in entities:
        line = f"doc_{e.document_id} [{e.entity_type}] {e.name}"
        if e.event_date:
            line += f" ({e.event_date})"
        line += f" — quote: '{e.raw_quote}'"
        lines.append(line)
    return "\n".join(lines)


async def run(
    *,
    briefing_id: UUID,
    user_id: UUID,
    target_specialty: Specialty,
) -> None:
    """Execute the 5-stage pipeline for a single briefing_id.

    Idempotent semantics: always updates the briefing row (status to ready or failed
    + completed_at always set in finally).
    """
    status = "failed"
    error_message: str | None = None
    payload: BriefingPayload | None = None
    html: str | None = None

    try:
        async with AsyncSessionLocal() as session:
            entities, sources = await _load_user_entities_and_sources(session, user_id)
            patient = await _load_user_patient_header(session, user_id)

        sections = format_sections(
            entities=entities,
            specialty=target_specialty,
            reason_for_consult=None,
        )

        template = load_template(target_specialty)
        entities_text = _entities_to_text_block(entities)
        qa_items, llm_meta = bake_questions(
            specialty=target_specialty,
            entities_text=entities_text,
            specialty_hints=template.qa_pre_baker_hints,
            source_raw_texts=sources,
        )
        sections.open_questions = qa_items

        payload = BriefingPayload(
            schema_version="1.0",
            generated_at=datetime.now(UTC),
            target_specialty=target_specialty,
            patient=patient,
            sections=sections,
            applied_filter=AppliedFilter(
                specialty=target_specialty.value, rules_version=template.rules_version
            ),
            llm_meta=llm_meta,
        )

        html = render_briefing_html(payload)
        status = "ready"

    except QAPrebakerError as e:
        error_message = f"qa_pre_baker failed: {e}"
        logger.error("briefing %s qa_pre_baker error: %s", briefing_id, e)
    except Exception as e:
        error_message = f"orchestrator error: {e}"
        logger.exception("briefing %s orchestrator unexpected error", briefing_id)

    # Always-write status update — even on failure, completed_at set
    try:
        async with AsyncSessionLocal() as session:
            await briefing_repo.update_payload_html_status(
                session,
                briefing_id=briefing_id,
                payload=payload,
                html_render=html,
                status=status,
                error_message=error_message,
            )
            await session.commit()
    except Exception as e:
        # Last-ditch logging; can't update DB
        logger.exception("briefing %s failed to write final status: %s", briefing_id, e)
```

```python
"""Briefings router — POST/GET/list/regenerate/DELETE endpoints.

Async generation pattern (per M5 brainstorm decision #1):
- POST returns 202 + briefing_id; orchestrator runs as BackgroundTask
- Frontend polls GET /briefings/{id} until status='ready'

Cross-user isolation: all routes use get_current_user_id Depends + 404 opaque.
"""
from __future__ import annotations

from uuid import UUID

from fastapi import APIRouter, BackgroundTasks, Depends, HTTPException, Response, status
from sqlalchemy.ext.asyncio import AsyncSession

from adelia.auth import get_current_user_id
from adelia.db import AsyncSessionLocal, get_db
from adelia.schemas import (
    BriefingCreate,
    BriefingDetailOut,
    BriefingOut,
    BriefingPayload,
    BriefingStatus,
    Specialty,
)
from adelia.services import briefing_orchestrator, briefing_repo

router = APIRouter(prefix="/briefings", tags=["briefings"])


def _to_out(briefing) -> BriefingOut:
    return BriefingOut(
        id=briefing.id,
        target_specialty=Specialty(briefing.target_specialty),
        status=BriefingStatus(briefing.status),
        is_stale=briefing.is_stale,
        created_at=briefing.created_at,
        completed_at=briefing.completed_at,
        error_message=briefing.error_message,
    )


def _to_detail(briefing) -> BriefingDetailOut:
    payload = None
    if briefing.payload:
        payload = BriefingPayload.model_validate(briefing.payload)
    return BriefingDetailOut(
        id=briefing.id,
        target_specialty=Specialty(briefing.target_specialty),
        status=BriefingStatus(briefing.status),
        is_stale=briefing.is_stale,
        created_at=briefing.created_at,
        completed_at=briefing.completed_at,
        error_message=briefing.error_message,
        payload=payload,
        html_render=briefing.html_render,
    )


def _enqueue(
    background_tasks: BackgroundTasks,
    briefing_id: UUID,
    user_id: UUID,
    target_specialty: Specialty,
) -> None:
    """Wrap orchestrator.run in BackgroundTasks. Override in tests."""
    background_tasks.add_task(
        briefing_orchestrator.run,
        briefing_id=briefing_id,
        user_id=user_id,
        target_specialty=target_specialty,
    )


@router.post("", status_code=status.HTTP_202_ACCEPTED, response_model=BriefingOut)
async def create_briefing(
    payload: BriefingCreate,
    background_tasks: BackgroundTasks,
    user_id: UUID = Depends(get_current_user_id),
    db: AsyncSession = Depends(get_db),
):
    briefing = await briefing_repo.create_briefing(
        db, user_id=user_id, target_specialty=payload.target_specialty
    )
    await db.commit()
    _enqueue(background_tasks, briefing.id, user_id, payload.target_specialty)
    return _to_out(briefing)


@router.get("", response_model=list[BriefingOut])
async def list_briefings(
    user_id: UUID = Depends(get_current_user_id),
    db: AsyncSession = Depends(get_db),
):
    rows = await briefing_repo.list_for_user(db, user_id=user_id)
    return [_to_out(b) for b in rows]


@router.get("/{briefing_id}", response_model=BriefingDetailOut)
async def get_briefing(
    briefing_id: UUID,
    user_id: UUID = Depends(get_current_user_id),
    db: AsyncSession = Depends(get_db),
):
    briefing = await briefing_repo.get_by_id(db, briefing_id=briefing_id, user_id=user_id)
    if briefing is None:
        raise HTTPException(status_code=404, detail="Not found")
    return _to_detail(briefing)


@router.post("/{briefing_id}/regenerate", status_code=status.HTTP_202_ACCEPTED, response_model=BriefingOut)
async def regenerate_briefing(
    briefing_id: UUID,
    background_tasks: BackgroundTasks,
    user_id: UUID = Depends(get_current_user_id),
    db: AsyncSession = Depends(get_db),
):
    original = await briefing_repo.get_by_id(db, briefing_id=briefing_id, user_id=user_id)
    if original is None:
        raise HTTPException(status_code=404, detail="Not found")
    new = await briefing_repo.create_briefing(
        db,
        user_id=user_id,
        target_specialty=Specialty(original.target_specialty),
    )
    await db.commit()
    _enqueue(background_tasks, new.id, user_id, Specialty(original.target_specialty))
    return _to_out(new)


@router.delete("/{briefing_id}", status_code=status.HTTP_204_NO_CONTENT)
async def delete_briefing(
    briefing_id: UUID,
    user_id: UUID = Depends(get_current_user_id),
    db: AsyncSession = Depends(get_db),
):
    ok = await briefing_repo.soft_delete(db, briefing_id=briefing_id, user_id=user_id)
    if not ok:
        raise HTTPException(status_code=404, detail="Not found")
    await db.commit()
    return Response(status_code=204)
```

```bash
echo "    doc_id=${DOC_ID}"

echo "==> [5/6] POST briefing endocrinology"
BRIEFING_ID=$(curl -fsS -X POST "${API_BASE}/briefings" \

    -H "Content-Type: application/json" \
```

```typescript
    const resp = await fetch(`${API_BASE}/briefings`, {
      headers: { Authorization: `Bearer ${token}` },
    });
    if (!resp.ok) throw new Error(`GET /briefings ${resp.status}`);
    return resp.json();

  async getById(id: string, token: string): Promise<BriefingDetailOut> {
    const resp = await fetch(`${API_BASE}/briefings/${id}`, {
      headers: { Authorization: `Bearer ${token}` },
    });
    if (!resp.ok) throw new Error(`GET /briefings/${id} ${resp.status}`);
    return resp.json();
  },

  async regenerate(id: string, token: string): Promise<BriefingOut> {
    const resp = await fetch(`${API_BASE}/briefings/${id}/regenerate`, {
      method: "POST",
      headers: { Authorization: `Bearer ${token}` },
    });
    if (!resp.ok) throw new Error(`POST /briefings/${id}/regenerate ${resp.status}`);
    return resp.json();
  },

  async remove(id: string, token: string): Promise<void> {
    const resp = await fetch(`${API_BASE}/briefings/${id}`, {
      method: "DELETE",
      headers: { Authorization: `Bearer ${token}` },
    });
    if (!resp.ok) throw new Error(`DELETE /briefings/${id} ${resp.status}`);
```

- [ ] **Step 5: Detail page with polling**

Create `apps/web/app/[locale]/dashboard/briefings/[id]/page.tsx`:

```tsx
"use client";

import { useCallback, useEffect, useState } from "react";
import { useRouter } from "next/navigation";
import { useTranslations } from "next-intl";

import { briefings as briefingsApi } from "@/lib/api-client";
import type { BriefingDetailOut } from "@/lib/briefing-types";
import { BriefingDetail } from "@/components/briefing/briefing-detail";
import { StaleBanner } from "@/components/briefing/stale-banner";

interface Props {
  params: Promise<{ locale: string; id: string }>;
}

export default function BriefingDetailPage({ params }: Props) {
  const t = useTranslations();
  const router = useRouter();
  const [briefing, setBriefing] = useState<BriefingDetailOut | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [regenerating, setRegenerating] = useState(false);
  const [resolvedParams, setResolvedParams] = useState<{ locale: string; id: string } | null>(null);

  useEffect(() => {
    params.then(setResolvedParams);
  }, [params]);

  const fetchBriefing = useCallback(async () => {
    if (!resolvedParams) return;
    try {
      const tokenResp = await fetch("/api/auth/token", { credentials: "include" });
      const { token } = await tokenResp.json();
      const b = await briefingsApi.getById(resolvedParams.id, token);
      setBriefing(b);
    } catch (e) {
      setError(String(e));
    }
  }, [resolvedParams]);

  useEffect(() => {
    fetchBriefing();
  }, [fetchBriefing]);

  // Poll every 2s while processing
  useEffect(() => {
    if (!briefing || briefing.status !== "processing") return;
    const interval = setInterval(fetchBriefing, 2000);
    return () => clearInterval(interval);
  }, [briefing, fetchBriefing]);

  async function handleRegenerate() {
    if (!resolvedParams) return;
    setRegenerating(true);
    try {
      const tokenResp = await fetch("/api/auth/token", { credentials: "include" });
      const { token } = await tokenResp.json();
      const newBriefing = await briefingsApi.regenerate(resolvedParams.id, token);
      router.push(`/${resolvedParams.locale}/dashboard/briefings/${newBriefing.id}`);
    } catch (e) {
      setError(String(e));
      setRegenerating(false);
    }
  }

  if (error) {
    return (
      <main className="mx-auto max-w-3xl px-6 py-10">
        <div className="rounded border border-red-300 bg-red-50 p-4 text-red-900">{error}</div>
      </main>
    );
  }

  if (!briefing) {
    return <main className="mx-auto max-w-3xl px-6 py-10">Loading...</main>;
  }

  return (
    <main className="mx-auto max-w-4xl px-6 py-10">
      <div className="mb-4 text-sm text-gray-600">
        Status: <strong>{t(`briefing.status.${briefing.status}`)}</strong>
        {briefing.status === "processing" && <span className="ml-2">— {t("briefing.polling")}</span>}
      </div>

      {briefing.is_stale && (
        <StaleBanner onRegenerate={handleRegenerate} disabled={regenerating} />
      )}

      {briefing.status === "failed" && (
        <div className="mb-4 rounded border border-red-300 bg-red-50 p-3 text-sm text-red-900">
          {t("briefing.errorBanner")} {briefing.error_message}
        </div>
      )}

      {briefing.status === "ready" && resolvedParams && (
        <BriefingDetail briefing={briefing} locale={resolvedParams.locale} />
      )}
    </main>
  );
}
```
