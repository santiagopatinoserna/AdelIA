# AdelIA v0.1 — M1 Foundation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the AdelIA repo skeleton with auth + file upload + OCR pipeline. End state: user signs in via magic link, drag-drops a PDF, sees status flip from "processing" to "ready", clicks the document and reads OCR'd text.

**Architecture:** Monorepo (pnpm workspaces + uv). Next.js 15 web app talks to FastAPI backend over REST. FastAPI uses BackgroundTasks (NO Celery in M1) for OCR. Files stored in S3-compatible storage (MinIO local dev / Cloudflare R2 prod). Postgres 16 with pgvector extension installed (only `documents` table populated in M1; `chunks`/`entities` tables created but unused until M2-M3). Auth.js v5 magic links via Resend.

**Tech Stack:**
- Backend: Python 3.12 + FastAPI 0.115 + uv + SQLAlchemy 2.0 + Alembic + boto3 + httpx + pydantic-settings + pytest + pytest-asyncio
- Frontend: Next.js 15 (App Router) + TypeScript 5 + Tailwind 4 + Auth.js v5 + next-intl 3 + Zod
- DB: Postgres 16 + pgvector
- Storage: MinIO 2024 (dev) / Cloudflare R2 (prod) via S3 API
- Email: Resend (BYOK API key)
- OCR: Mistral OCR API (BYOK API key)
- Containers: Docker Compose
- Package managers: pnpm 9 (frontend) + uv 0.4 (backend)

**Out of scope for M1 (deferred to M2+):** audio transcription, entity extraction (LLM structured), embeddings/pgvector search, RAG chat (Modo A), briefing generator (Modo B), sharing/PDF, EN+FR strings, Telegram/email/PWA ingestion, Celery, Fly.io deploy, observability stack.

**Time estimate (1h/day):** ~20-30h work, ~3-5 weeks at 1h/day with Claude Code leverage.

**Acceptance test (M1 done):** From a fresh clone, run `docker compose up -d` then `pnpm dev` then `uv run uvicorn adelia.main:app --reload`. Open `http://localhost:3000/es`, click "Sign in", receive magic link, drag-drop `tests/fixtures/sample-lab-result.pdf`, see "processing" badge for ~5s, then "ready", click document, see extracted text matching the PDF content.

```python
# apps/api/alembic/env.py
import asyncio
from logging.config import fileConfig

from sqlalchemy import pool
from sqlalchemy.engine import Connection
from sqlalchemy.ext.asyncio import async_engine_from_config

from alembic import context
from adelia.db import Base
from adelia import models  # noqa: F401 — register tables
from adelia.config import settings

config = context.config
config.set_main_option(
    "sqlalchemy.url",
    settings.database_url.replace("+asyncpg", ""),
)

if config.config_file_name is not None:
    fileConfig(config.config_file_name)

target_metadata = Base.metadata


def do_run_migrations(connection: Connection) -> None:
    context.configure(connection=connection, target_metadata=target_metadata)
    with context.begin_transaction():
        context.run_migrations()


async def run_async_migrations() -> None:
    connectable = async_engine_from_config(
        config.get_section(config.config_ini_section) or {},
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
        url=settings.database_url,
    )
    async with connectable.connect() as connection:
        await connection.run_sync(do_run_migrations)
    await connectable.dispose()


def run_migrations_online() -> None:
    asyncio.run(run_async_migrations())


run_migrations_online()
```

```python
# apps/api/adelia/storage.py
import asyncio
from typing import BinaryIO

import boto3
from botocore.client import Config

from adelia.config import settings


class S3Storage:
    def __init__(self) -> None:
        self._client = boto3.client(
            "s3",
            endpoint_url=settings.s3_endpoint_url,
            aws_access_key_id=settings.s3_access_key,
            aws_secret_access_key=settings.s3_secret_key,
            region_name=settings.s3_region,
            config=Config(signature_version="s3v4"),
        )
        self._bucket = settings.s3_bucket

    async def put_object(self, key: str, body: BinaryIO, content_type: str) -> None:
        await asyncio.to_thread(
            self._client.put_object,
            Bucket=self._bucket,
            Key=key,
            Body=body,
            ContentType=content_type,
        )

    async def get_object(self, key: str) -> BinaryIO:
        response = await asyncio.to_thread(
            self._client.get_object, Bucket=self._bucket, Key=key
        )
        return response["Body"]

    async def delete_object(self, key: str) -> None:
        await asyncio.to_thread(
            self._client.delete_object, Bucket=self._bucket, Key=key
        )

    async def generate_presigned_url(self, key: str, expires_in: int = 3600) -> str:
        return await asyncio.to_thread(
            self._client.generate_presigned_url,
            "get_object",
            Params={"Bucket": self._bucket, "Key": key},
            ExpiresIn=expires_in,
        )
```

```python
# apps/api/adelia/auth.py
from typing import Annotated
from uuid import UUID

from fastapi import Depends, HTTPException, status
from fastapi.security import HTTPAuthorizationCredentials, HTTPBearer
from jose import JWTError, jwt

from adelia.config import settings


bearer_scheme = HTTPBearer()


def decode_session_token(token: str) -> dict:
    return jwt.decode(token, settings.auth_secret, algorithms=["HS256"])


async def get_current_user_id(
    credentials: Annotated[HTTPAuthorizationCredentials, Depends(bearer_scheme)],
) -> UUID:
    try:
        payload = decode_session_token(credentials.credentials)
    except JWTError:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid token",
            headers={"WWW-Authenticate": "Bearer"},
        )
    sub = payload.get("sub")
    if not sub:
        raise HTTPException(status_code=401, detail="Token missing sub")
    return UUID(sub)
```

```python
# apps/api/adelia/workers/ocr.py
import base64

import httpx

from adelia.config import settings


MISTRAL_OCR_URL = "https://api.mistral.ai/v1/ocr"


async def extract_text_from_pdf(body: bytes, mime_type: str) -> str:
    """Send the file to Mistral OCR API and return concatenated text from all pages."""
    if not settings.mistral_api_key:
        raise RuntimeError("MISTRAL_API_KEY not set")

    encoded = base64.b64encode(body).decode("ascii")
    headers = {
        "Authorization": f"Bearer {settings.mistral_api_key}",
        "Content-Type": "application/json",
    }
    payload = {
        "model": "mistral-ocr-latest",
        "document": {
            "type": "document_base64",
            "document_base64": encoded,
            "document_name": f"upload.{_ext_from_mime(mime_type)}",
        },
        "include_image_base64": False,
    }

    async with httpx.AsyncClient(timeout=120.0) as client:
        response = await client.post(MISTRAL_OCR_URL, headers=headers, json=payload)
        response.raise_for_status()
        data = response.json()

    pages = data.get("pages", [])
    return "\n\n".join(page.get("markdown", "") for page in pages)


def _ext_from_mime(mime_type: str) -> str:
    return {
        "application/pdf": "pdf",
        "image/png": "png",
        "image/jpeg": "jpg",
    }.get(mime_type, "bin")
```

```python
# apps/api/adelia/routers/documents.py
import uuid
from datetime import datetime, timezone
from typing import Annotated

from fastapi import APIRouter, BackgroundTasks, Depends, File, HTTPException, UploadFile, status
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from adelia.auth import get_current_user_id
from adelia.db import AsyncSessionLocal, get_db
from adelia.models import Document
from adelia.schemas import DocumentOut, derive_status
from adelia.storage import S3Storage
from adelia.workers.ocr import extract_text_from_pdf


router = APIRouter(prefix="/documents", tags=["documents"])
storage = S3Storage()

ALLOWED_MIME = {"application/pdf", "image/png", "image/jpeg", "audio/mpeg", "audio/wav"}


def _doc_to_out(doc: Document) -> DocumentOut:
    return DocumentOut(
        id=doc.id,
        source=doc.source,
        mime_type=doc.mime_type,
        s3_key=doc.s3_key,
        raw_text=doc.raw_text,
        created_at=doc.created_at,
        processed_at=doc.processed_at,
        status=derive_status(doc.raw_text, doc.processed_at),
    )


async def process_document(document_id: uuid.UUID, mime_type: str) -> None:
    """BackgroundTask: download from S3, OCR, persist raw_text, mark processed_at."""
    async with AsyncSessionLocal() as session:
        doc = await session.get(Document, document_id)
        if doc is None:
            return
        try:
            body_stream = await storage.get_object(doc.s3_key)
            body = body_stream.read()
            if mime_type.startswith("application/pdf") or mime_type.startswith("image/"):
                text = await extract_text_from_pdf(body, mime_type)
            else:
                text = ""  # audio handled in M2
            doc.raw_text = text
        except Exception:
            doc.raw_text = None
        finally:
            doc.processed_at = datetime.now(timezone.utc)
            await session.commit()


@router.post("", response_model=DocumentOut, status_code=status.HTTP_201_CREATED)
async def upload_document(
    background: BackgroundTasks,
    user_id: Annotated[uuid.UUID, Depends(get_current_user_id)],
    db: Annotated[AsyncSession, Depends(get_db)],
    file: UploadFile = File(...),
):
    if file.content_type not in ALLOWED_MIME:
        raise HTTPException(status_code=415, detail=f"Unsupported mime: {file.content_type}")

    body = await file.read()
    s3_key = f"users/{user_id}/{uuid.uuid4()}-{file.filename}"
    import io
    await storage.put_object(s3_key, io.BytesIO(body), content_type=file.content_type)

    doc = Document(
        user_id=user_id,
        source="web",
        mime_type=file.content_type,
        s3_key=s3_key,
    )
    db.add(doc)
    await db.commit()
    await db.refresh(doc)

    background.add_task(process_document, doc.id, file.content_type)

    return _doc_to_out(doc)


@router.get("", response_model=list[DocumentOut])
async def list_documents(
    user_id: Annotated[uuid.UUID, Depends(get_current_user_id)],
    db: Annotated[AsyncSession, Depends(get_db)],
):
    result = await db.execute(
        select(Document).where(Document.user_id == user_id, Document.deleted_at.is_(None))
        .order_by(Document.created_at.desc())
    )
    docs = result.scalars().all()
    return [_doc_to_out(d) for d in docs]


@router.get("/{doc_id}", response_model=DocumentOut)
async def get_document(
    doc_id: uuid.UUID,
    user_id: Annotated[uuid.UUID, Depends(get_current_user_id)],
    db: Annotated[AsyncSession, Depends(get_db)],
):
    doc = await db.get(Document, doc_id)
    if doc is None or doc.user_id != user_id or doc.deleted_at is not None:
        raise HTTPException(status_code=404)
    return _doc_to_out(doc)
```

```ts
import { auth } from "@/lib/auth";

const API_URL = process.env.NEXT_PUBLIC_API_URL ?? "http://localhost:8000";

export type DocumentStatus = "uploaded" | "processing" | "ready" | "failed";

export type DocumentDTO = {
  id: string;
  source: "web" | "manual";
  mime_type: string;
  s3_key: string;
  raw_text: string | null;
  created_at: string;
  processed_at: string | null;
  status: DocumentStatus;
};

async function authedFetch(path: string, init?: RequestInit) {
  const session = await auth();
  const token = (session as { jwt?: string } | null)?.jwt;
  const headers = new Headers(init?.headers);
  if (token) headers.set("Authorization", `Bearer ${token}`);
  return fetch(`${API_URL}${path}`, { ...init, headers, cache: "no-store" });
}

export async function listDocuments(): Promise<DocumentDTO[]> {
  const res = await authedFetch("/documents");
  if (!res.ok) throw new Error(`listDocuments: ${res.status}`);
  return res.json();
}

export async function getDocument(id: string): Promise<DocumentDTO> {
  const res = await authedFetch(`/documents/${id}`);
  if (!res.ok) throw new Error(`getDocument: ${res.status}`);
  return res.json();
}
```

```tsx
// apps/web/components/upload-zone.tsx
"use client";
import { useState } from "react";
import { useRouter } from "next/navigation";
import { useTranslations } from "next-intl";

export function UploadZone({ apiUrl, jwt, locale }: { apiUrl: string; jwt: string; locale: string }) {
  const t = useTranslations("upload");
  const router = useRouter();
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState<string | null>(null);

  async function upload(file: File) {
    setBusy(true);
    setError(null);
    try {
      const fd = new FormData();
      fd.append("file", file);
      const res = await fetch(`${apiUrl}/documents`, {
        method: "POST",
        body: fd,
        headers: { Authorization: `Bearer ${jwt}` },
      });
      if (!res.ok) throw new Error(`status ${res.status}`);
      router.push(`/${locale}/dashboard`);
      router.refresh();
    } catch (e) {
      setError(String(e));
    } finally {
      setBusy(false);
    }
  }

  return (
    <div
      onDragOver={(e) => e.preventDefault()}
      onDrop={(e) => {
        e.preventDefault();
        const f = e.dataTransfer.files[0];
        if (f) upload(f);
      }}
      className="rounded-lg border-2 border-dashed border-slate-300 bg-white px-6 py-16 text-center"
    >
      <p className="text-slate-600">{t("drop")}</p>
      <p className="mt-1 text-sm text-slate-400">{t("or")}</p>
      <label className="mt-4 inline-block cursor-pointer rounded-md bg-slate-900 px-4 py-2 text-sm text-white">
        {t("browse")}
        <input
          type="file"
          accept="application/pdf,image/png,image/jpeg,audio/*"
          className="hidden"
          onChange={(e) => {
            const f = e.target.files?.[0];
            if (f) upload(f);
          }}
        />
      </label>
      {busy && <p className="mt-4 text-amber-700">{t("uploading")}</p>}
      {error && <p className="mt-4 text-rose-700">{error}</p>}
    </div>
  );
}
```
